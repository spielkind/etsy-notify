99 def event_settings(self) -> EventSettings: 

100 """The events to notify, with ``NOTIFY_BACK=false`` removing "back".""" 

101 enabled = {e.strip() for e in self.notify_events.split(",") if e.strip()} 

102 if not self.notify_back: 

103 enabled.discard("back") 

104 return EventSettings( 

105 enabled=frozenset(enabled), 

106 price_change_percent=self.notify_price_change_percent, 

107 low_stock=self.notify_low_stock, 

108 )
